log_level=DEBUG  # TRACE, DEBUG, INFO, WARN, ERROR
filter_sampling_params=true  # Drops top_k, temperature and top_p whatever their values. Anthropic deprecated all three on the current models and answers 400 "`x` is deprecated for this model" before a token is generated: top_k on every Claude 5, temperature and top_p from claude-opus-4-7 onward. Front ends send them on every request, so leaving this off makes every newer model fail before it starts. Set to false to keep sampling control on claude-opus-4-6 and older, which still honour it.
upstream_timeout_ms=300000  # Abort a request to api.anthropic.com after this many ms of socket inactivity. Streaming resets the timer.
inject_billing_header=true  # Send the billing header block and metadata.user_id the way Claude Code 2.1.280 does. Both are derived from your token, never randomized, so the cached prefix stays byte-identical between turns. Set to false if Anthropic ever rejects a system array that does not open with the Claude Code sentence.
cloak_tools=true  # Rename the tools your front end declares (suffix _ide) and declare Claude Code's own tool set alongside them, marked unavailable, so the tool list looks like the CLI that owns the token. Names are put back in the response, streaming included. Set to false if your front end trips over it.
redact_thinking=false  # On means the beta list matches the CLI byte for byte. Costs only the reasoning text: measured against api.anthropic.com with the proxy out of the path, the same prompt spends the same thinking tokens either way (39 on claude-opus-4-6) and the signature still comes back, so the model thinks exactly as much and only the text is withheld. Set to false if your front end displays reasoning and you want to read it.
strip_cache_control_ttl=false  # Leave false. Upstream accepts cache_control.ttl, so stripping it downgrades every "1h" breakpoint your front end asks for to the 5m default. Set to true only if Anthropic starts rejecting the field.

proxy_api_key=  # Blank means anyone who can reach this port spends your subscription. Set a secret and clients must send it as `Authorization: Bearer <key>` or `x-api-key: <key>` on /v1/*. The PROXY_API_KEY env var overrides this. /auth/* still works from localhost without the key so the browser login flow keeps working; from anywhere else it needs the key too.
auth_routes_enabled=true  # Set to false to stop serving /auth/* entirely: login, callback, status and logout all answer 404, and the proxy handles /v1 traffic only. Use it once the proxy is authenticated (or gets its credentials from a mounted ~/.claude), so nothing reachable over the network can replace or drop the tokens it holds. The AUTH_ROUTES_ENABLED env var overrides this. Turn it back on and restart when you need to log in again.
